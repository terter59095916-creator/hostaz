<script>
(function() {
  let knWs = null, knMyId = null, knActiveStreamId = null, knTimerInterval = null;
  function knConnect() {
    const proto = location.protocol === 'https:' ? 'wss:' : 'ws:';
    knWs = new WebSocket(proto + '//' + location.host + '/ws/');
    knWs.binaryType = 'arraybuffer';
    knWs.onmessage = (e) => {
      try {
        const buf = new Uint8Array(e.data);
        const len = buf[0] * 256 + buf[1];
        const jsonStr = new TextDecoder('utf-8').decode(buf.slice(2, 2 + len));
        const msg = JSON.parse(jsonStr);
        knHandleMessage(msg);
      } catch (err) {}
    };
    knWs.onclose = () => { setTimeout(knConnect, 3000); };
  }
  function knSend(obj) {
    if (!knWs || knWs.readyState !== WebSocket.OPEN) return;
    const utf8 = new TextEncoder().encode(JSON.stringify(obj));
    const framed = new Uint8Array(2 + utf8.length);
    framed[0] = Math.floor(utf8.length / 256); framed[1] = utf8.length % 256;
    framed.set(utf8, 2);
    knWs.send(framed);
  }
  function knHandleMessage(msg) {
    if (msg.type === 'user_kickout') {
      fetch('/api/profile/me', { headers: { 'Authorization': 'Bearer ' + localStorage.getItem('authToken') } })
        .then(r => r.json()).then(me => {
          if (String(me.id) === String(msg.kicked_user.id)) {
            knShowNotify(msg.kicker_user.name, msg.kicked_user.id, msg.game_id);
          }
        }).catch(() => {});
    } else if (msg.type === 'user_save' || msg.type === 'user_kicked') {
      knHide();
    }
  }
  window.knMyGameId = null;
  function knShowNotify(kickerName, myUserId, gameId) {
    knMyId = myUserId; knMyGameId = gameId;
    document.getElementById('kn-kicker-name').textContent = kickerName;
    document.getElementById('kickout-notify-overlay').classList.add('show');
    let t = 30;
    document.getElementById('kn-timer').textContent = t;
    if (knTimerInterval) clearInterval(knTimerInterval);
    knTimerInterval = setInterval(() => {
      t--;
      document.getElementById('kn-timer').textContent = Math.max(0, t);
      if (t <= 0) { clearInterval(knTimerInterval); knHide(); }
    }, 1000);
  }
  function knHide() {
    document.getElementById('kickout-notify-overlay').classList.remove('show');
    if (knTimerInterval) clearInterval(knTimerInterval);
  }
  window.knPayToStay = function() {
    knSend({ type: 'user_save', user_id: knMyId, save_referrer: 'self' });
    knHide();
  };
  window.knDismiss = function() { knHide(); };
  knConnect();
})();
</script>
